```typescript
/**
 * @file questionRouter.ts
 * @description Step 16c: LLM-as-judge dynamic router for OM config selection.
 *
 * Classifies each LongMemEval-style question into one of six categories,
 * then routes to the per-category-optimal OM configuration based on the
 * empirical 7-variant scoreboard from Steps 15 v1-v5 and Step 16.
 *
 * The classifier runs a cheap gpt-5-mini call per question (~$0.0005).
 * Routing is a pure lookup. See
 * docs/specs/2026-04-22-step-16c-llm-judge-router-design.md for the
 * routing table and rationale.
 */

import type { IReader } from '../readers/IReader.js';

export type QuestionCategory =
  | 'single-session-user'
  | 'single-session-assistant'
  | 'single-session-preference'
  | 'knowledge-update'
  | 'multi-session'
  | 'temporal-reasoning';

const CATEGORY_TOKENS: readonly QuestionCategory[] = [
  'single-session-user',
  'single-session-assistant',
  'single-session-preference',
  'knowledge-update',
  'multi-session',
  'temporal-reasoning',
];

export const CLASSIFIER_SYSTEM_PROMPT = `You are classifying a memory-system question into one of six categories.

Return ONLY the category token (no explanation, no quotes, no punctuation).

Categories:
- single-session-user: the question asks about something the USER said, did, or stated in a specific past session. Answer is in one session. Examples: "What did I tell you about my favorite dessert?", "Where did I say I moved to last month?"
- single-session-assistant: the question asks about something the ASSISTANT said, generated, or recommended in a specific session. Answer is in one session. Examples: "What recipe did you suggest for the birthday party?", "What books did you recommend to me?"
- single-session-preference: the question asks about a preference the user stated in passing. Answer is in one session. Examples: "Do I prefer tea or coffee?", "What's my favorite type of movie?"
- knowledge-update: the question asks about current state where the answer EVOLVED across sessions (supersession). Examples: "What's my current job title?", "Where do I live now?", "What's my latest project?"
- multi-session: the question requires combining information from 2+ separate sessions. Examples: "How many different languages have I mentioned studying?", "Which authors did you recommend across our conversations?"
- temporal-reasoning: the question asks about the order, timing, or duration of events across time. Examples: "In what order did I visit the three countries?", "How many months ago did I start the new job?"`;

export interface ClassifyQuestionArgs {
  reader: IReader;
  question: string;
}

export async function classifyQuestion(args: ClassifyQuestionArgs): Promise<QuestionCategory> {
  const { reader, question } = args;
  const userPrompt = `Question: ${question}\n\nCategory:`;
  const response = await reader.invoke({
    system: CLASSIFIER_SYSTEM_PROMPT,
    user: userPrompt,
    maxTokens: 16,
    temperature: 0,
  });
  const raw = response.text.trim().toLowerCase();
  for (const token of CATEGORY_TOKENS) {
    if (raw === token || raw.startsWith(`${token} `) || raw.startsWith(`${token}\n`)) {
      return token;
    }
  }
  // Default to multi-session for unparseable output — hardest category
  // and the one that benefits most from retrieval+OM combined.
  return 'multi-session';
}

export interface RouterConfig {
  /** True = run OM ingest (gpt-4o observer + dual-path). False = skip OM, use canonical Step 3 retrieval. */
  runOM: boolean;
  /** True = run HybridRetriever (BM25 + dense + RRF + Cohere rerank). Always true in v2 routing. */
  runRetrieval: boolean;
  /** True = include cross-session reader scaffold. Only consulted when runOM=true. */
  useScaffold: boolean;
}

export const DEFAULT_ROUTE: RouterConfig = {
  runOM: true,
  runRetrieval: false,
  useScaffold: true,
};
// Default falls back to v5 OM (with scaffold) on classifier errors. OM at least
// covers MS via cross-session synthesis; canonical with no OM would miss MS
// questions if they're misclassified.

export function routeQuestion(category: QuestionCategory): RouterConfig {
  switch (category) {
    case 'single-session-user':
    case 'single-session-assistant':
    case 'single-session-preference':
    case 'temporal-reasoning':
      // Phase A v5 N=102 data: canonical wins these 4 categories.
      // SSU 97.1% vs v5 94.1%; SSA 89.3% vs v5 76.5%; SSP 60.0% vs v5 58.8%; TR 70.2% vs v5 52.9%.
      // Skip OM ingest entirely; use Step 3 HybridRetriever path.
      return { runOM: false, runRetrieval: true, useScaffold: false };
    case 'knowledge-update':
    case 'multi-session':
      // Phase A v5 N=102 data: v5 wins these 2 categories.
      // KU 88.2% vs canonical 86.8% (+1.4pp tied within noise).
      // MS 70.6% vs canonical 54.9% (+15.7pp — load-bearing routing decision).
      // Run OM ingest with scaffold; no retrieval (Step 16 family was RED).
      return { runOM: true, runRetrieval: false, useScaffold: true };
  }
}
```

- [ ] **Step 2: Branch BEFORE OM ingest (v2 architecture change)**

The v2 router decides per-case whether to run OM at all. If `runOM === false`, skip the entire OM ingest and fall through to the canonical Step 3 HybridRetriever path. This is a different shape from v1 (which always ran OM and only varied the reader prompt).

In `runCase`, BEFORE the existing `if (this.opts.observationalMemoryEnabled)` block:

```typescript
// Step 16c v2: classify question + decide route BEFORE branching to OM or canonical.
let routerCategory: QuestionCategory | null = null;
let route: RouterConfig | null = null;
const dynamicRouterOn = Boolean(this.opts.observationalMemoryEnabled && ctx.config.omDynamicRouter);

if (dynamicRouterOn) {
  const classifierReader = createReader('gpt-5-mini');
  try {
    routerCategory = await classifyQuestion({
      reader: classifierReader,
      question: c.question,
    });
    route = routeQuestion(routerCategory);
    ctx.costTracker.record({
      category: 'reader',
      model: 'gpt-5-mini',
      tokensIn: 120,
      tokensOut: 4,
      tag: `${c.id}:om-classifier`,
    });
  } catch {
    // Classifier failure: fall back to v5 OM (DEFAULT_ROUTE).
    routerCategory = 'multi-session';
    route = DEFAULT_ROUTE;
  }
}

// Decide branch:
//   - dynamicRouterOn AND route.runOM === false: canonical path (skip OM)
//   - dynamicRouterOn AND route.runOM === true:  OM path with route.useScaffold
//   - dynamicRouterOn === false AND OM enabled: existing OM path (no router)
//   - dynamicRouterOn === false AND OM disabled: existing canonical path
const useOmBranch = dynamicRouterOn ? route!.runOM : Boolean(this.opts.observationalMemoryEnabled);

if (!useOmBranch) {
  // Canonical Step 3 HybridRetriever path. Same logic as the existing
  // non-OM branch, but record router metadata if dynamicRouterOn.
  const result = await this.runFullCognitiveCase(c, ctx);
  if (dynamicRouterOn) {
    result.metadata = {
      ...result.metadata,
      omDynamicRouter: true,
      omRouterCategory: routerCategory,
      omRouterRunOM: false,
      omRouterScaffold: false,
      omRouterRetrieval: true,
    };
  }
  return result;
}
```
